  guaMap() {
    const m = this.state.mobile;
    const { guaIdx, guaHover, guaLayer, pocIdx, pocHover } = this.state;
    const pinsOn = guaLayer === "utama";
    const open = guaIdx !== null;
    const g = open ? GUAS[guaIdx] : null;
    const p = pocIdx !== null ? POCKETS[pocIdx] : null;
    const go = i => this.setState({ guaIdx: (i + GUAS.length) % GUAS.length });
    const askFor = name => () => {
      this.setState({ guaIdx: null, pocIdx: null, dest: name });
      setTimeout(() => {
        const el = document.getElementById("form-ziarah");
        if (el) window.scrollTo({ top: el.getBoundingClientRect().top + window.pageYOffset - 90, behavior: "smooth" });
      }, 40);
    };

    return {
      // Yogyakarta punya empat titik dalam radius 25 km. Di bawah ~1180px
      // lebar peta, jaraknya jatuh di bawah lebar bulatan dan penanda saling
      // menelan area klik — jadi peta punya lebar minimum dan digeser, bukan
      // dikecilkan.
      mapScrollStyle: "overflow-x:auto;overflow-y:hidden;-webkit-overflow-scrolling:touch;" + (m
        ? "margin-top:22px;padding:0 20px 4px"
        : "margin-top:26px;padding:0 20px 4px"),
      mapInnerStyle: `position:relative;margin:0 auto;${m ? "width:1250px" : "width:100%;min-width:1180px;max-width:1600px"}`,
      guas: GUAS.map((x, i) => {
        const on = guaIdx === i || guaHover === i;
        const sel = guaIdx === i;
        const anchor = x.x > 78 ? "right:50%;transform:translateX(15px)"
          : x.x < 14 ? "left:50%;transform:translateX(-15px)"
          : "left:50%;transform:translateX(-50%)";
        return {
          no: x.no, short: x.short,
          aria: `${x.name}, ${x.area}`,
          onSelect: () => this.setState({ guaIdx: i }),
          onEnter: () => { if (!m) this.setState({ guaHover: i }); },
          onLeave: () => { if (!m) this.setState({ guaHover: null }); },
          hitStyle: `position:absolute;left:${x.x}%;top:${x.y}%;transform:translate(-50%,-50%);width:32px;height:32px;display:${pinsOn ? "flex" : "none"};align-items:center;justify-content:center;padding:0;border:none;background:none;z-index:${on ? 8 : 4}`,
          dotStyle: `width:${m ? 24 : 26}px;height:${m ? 24 : 26}px;border-radius:50%;display:flex;align-items:center;justify-content:center;font-family:'Space Mono',monospace;font-weight:700;font-size:11px;transition:transform .18s ease,background .18s ease,box-shadow .18s ease;` + (sel
            ? "background:#C9E45B;color:#1A1712;transform:scale(1.22);box-shadow:0 0 0 5px rgba(201,228,91,0.32),0 4px 14px rgba(26,23,18,0.4)"
            : on
              ? "background:#F7F3EA;color:#6C4BA6;transform:scale(1.14);box-shadow:0 0 0 3px rgba(247,243,234,0.5),0 0 0 2px #6C4BA6 inset,0 4px 12px rgba(26,23,18,0.32)"
              : "background:#F7F3EA;color:#6C4BA6;box-shadow:0 0 0 2px rgba(108,75,166,0.85),0 2px 7px rgba(26,23,18,0.26)"),
          labelStyle: `position:absolute;top:calc(50% + 21px);${anchor};white-space:nowrap;padding:5px 10px;border-radius:999px;background:#1A1712;color:#F7F3EA;font-family:'Space Mono',monospace;font-size:10px;letter-spacing:0.06em;pointer-events:none;transition:opacity .16s ease;opacity:${on ? 1 : 0}`,
          chipStyle: `flex:none;padding:9px 15px;border-radius:999px;font-size:11px;letter-spacing:0.06em;white-space:nowrap;transition:background .16s ease,color .16s ease;` + (sel
            ? "border:1px solid #1A1712;background:#1A1712;color:#F7F3EA"
            : "border:1px solid #D6CBB8;background:transparent;color:#5B5347")
        };
      }),
      swipeHintStyle: this.state.mapScrolls
        ? "display:block;text-align:center;padding:10px 20px 0;font-family:'Space Mono',monospace;font-size:10px;letter-spacing:0.1em;color:#9A9083"
        : "display:none",
      railStyle: m
        ? "display:flex;gap:8px;overflow-x:auto;padding:18px 20px 6px;-webkit-overflow-scrolling:touch"
        : "display:flex;flex-wrap:wrap;gap:8px;max-width:1280px;margin:0 auto;padding:24px 28px 0",
      pockets: POCKETS.map((x, i) => {
        const on = pocIdx === i || pocHover === i;
        const sel = pocIdx === i;
        const n = x.items.length;
        const size = Math.round(Math.min(50, 27 + n * 1.9));
        const anchor = x.x > 78 ? "right:50%;transform:translateX(15px)"
          : x.x < 14 ? "left:50%;transform:translateX(-15px)"
          : "left:50%;transform:translateX(-50%)";
        return {
          count: String(n), short: x.short,
          aria: `${x.region} — ${n} titik ziarah`,
          onSelect: () => this.setState({ pocIdx: i }),
          onEnter: () => { if (!m) this.setState({ pocHover: i }); },
          onLeave: () => { if (!m) this.setState({ pocHover: null }); },
          hitStyle: `position:absolute;left:${x.x}%;top:${x.y}%;transform:translate(-50%,-50%);width:${size + 8}px;height:${size + 8}px;display:${pinsOn ? "none" : "flex"};align-items:center;justify-content:center;padding:0;border:none;background:none;z-index:${on ? 8 : 4}`,
          dotStyle: `width:${size}px;height:${size}px;border-radius:50%;display:flex;align-items:center;justify-content:center;font-family:'Space Mono',monospace;font-weight:700;font-size:${size > 38 ? 14 : 12}px;transition:transform .18s ease,background .18s ease,box-shadow .18s ease;` + (sel
            ? "background:#C9E45B;color:#1A1712;transform:scale(1.12);box-shadow:0 0 0 5px rgba(201,228,91,0.3),0 0 0 14px rgba(201,228,91,0.14),0 4px 14px rgba(26,23,18,0.36)"
            : on
              ? "background:#F7F3EA;color:#6C4BA6;transform:scale(1.08);box-shadow:0 0 0 2px #6C4BA6,0 0 0 13px rgba(108,75,166,0.2),0 4px 12px rgba(26,23,18,0.3)"
              : "background:rgba(247,243,234,0.95);color:#6C4BA6;box-shadow:0 0 0 2px rgba(108,75,166,0.8),0 0 0 10px rgba(108,75,166,0.15),0 2px 7px rgba(26,23,18,0.22)"),
          labelStyle: `position:absolute;top:calc(50% + ${Math.round(size / 2) + 14}px);${anchor};white-space:nowrap;padding:5px 10px;border-radius:999px;background:#1A1712;color:#F7F3EA;font-family:'Space Mono',monospace;font-size:10px;letter-spacing:0.06em;pointer-events:none;transition:opacity .16s ease;opacity:${on ? 1 : 0}`
        };
      }),
      chips: pinsOn
        ? GUAS.map((x, i) => ({
            label: `${x.no} · ${x.short}`,
            pick: () => this.setState({ guaIdx: i }),
            style: `flex:none;padding:9px 15px;border-radius:999px;font-size:11px;letter-spacing:0.06em;white-space:nowrap;transition:background .16s ease,color .16s ease;` + (guaIdx === i
              ? "border:1px solid #1A1712;background:#1A1712;color:#F7F3EA"
              : "border:1px solid #D6CBB8;background:transparent;color:#5B5347")
          }))
        : POCKETS.map((x, i) => ({
            label: `${x.short} · ${x.items.length}`,
            pick: () => this.setState({ pocIdx: i }),
            style: `flex:none;padding:9px 15px;border-radius:999px;font-size:11px;letter-spacing:0.06em;white-space:nowrap;transition:background .16s ease,color .16s ease;` + (pocIdx === i
              ? "border:1px solid #1A1712;background:#1A1712;color:#F7F3EA"
              : "border:1px solid #D6CBB8;background:transparent;color:#5B5347")
          })),
      layers: [
        { id: "utama", label: `${GUAS.length} TITIK UTAMA` },
        { id: "semua", label: `SEMUA KANTONG · ${POCKET_TOTAL}` }
      ].map(t => {
        const on = guaLayer === t.id;
        return {
          label: t.label,
          pick: () => this.setState({ guaLayer: t.id, guaIdx: null, pocIdx: null, guaHover: null, pocHover: null }),
          style: `padding:10px 16px;border-radius:999px;border:none;font-size:11px;letter-spacing:0.08em;white-space:nowrap;background:${on ? "#1A1712" : "transparent"};color:${on ? "#F7F3EA" : "rgba(26,23,18,0.6)"}`
        };
      }),
      layerNote: pinsOn
        ? "Titik-titik yang paling sering diminta rombongan."
        : "Penanda menunjukkan jumlah titik per wilayah, bukan posisi masing-masing gua.",
      pocOpen: pocIdx !== null,
      poc: p ? {
        region: p.region, dio: p.dio, count: String(p.items.length),
        ask: askFor(`Kantong ziarah ${p.region}`),
        items: p.items.map(([name, place]) => ({
          name, place,
          no: MAJOR[name] || "",
          badgeStyle: MAJOR[name]
            ? "flex:none;width:26px;height:26px;border-radius:50%;display:flex;align-items:center;justify-content:center;background:#C9E45B;color:#1A1712;font-family:'Space Mono',monospace;font-weight:700;font-size:10px"
            : "flex:none;width:26px;height:26px;border-radius:50%;display:flex;align-items:center;justify-content:center;border:1px solid #D6CBB8;background:transparent",
          ask: askFor(name)
        }))
      } : { region: "", dio: "", count: "", ask: () => {}, items: [] },
      guaOpen: open,
      closeGua: () => this.setState({ guaIdx: null, pocIdx: null }),
      prevGua: () => go(guaIdx - 1),
      nextGua: () => go(guaIdx + 1),
      sheetStyle: m
        ? "position:fixed;left:0;right:0;bottom:0;z-index:89;background:#F7F3EA;border-radius:26px 26px 0 0;padding:10px 22px 28px;max-height:78vh;overflow-y:auto;box-shadow:0 -18px 50px rgba(26,23,18,0.24)"
        : "position:fixed;top:0;right:0;bottom:0;width:min(430px,92vw);z-index:89;background:#F7F3EA;padding:38px 34px;overflow-y:auto;box-shadow:-18px 0 54px rgba(26,23,18,0.2)",
      handleStyle: m
        ? "width:46px;height:4px;border-radius:999px;background:#D6CBB8;margin:0 auto 18px"
        : "display:none",
      sheetCloseStyle: `position:absolute;top:${m ? 14 : 22}px;right:${m ? 16 : 24}px;width:40px;height:40px;border:none;background:none;font-family:'Space Mono',monospace;font-size:24px;line-height:1;color:#1A1712`,
      gua: g ? {
        no: g.no, name: g.name, area: g.area, note: g.note, tags: g.tags,
        ask: () => {
          this.setState({ guaIdx: null, dest: g.name });
          setTimeout(() => {
            const el = document.getElementById("form-ziarah");
            if (el) window.scrollTo({ top: el.getBoundingClientRect().top + window.pageYOffset - 90, behavior: "smooth" });
          }, 40);
        }
      } : { no: "", name: "", area: "", note: "", tags: [], ask: () => {} }
    };
